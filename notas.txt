1. Preparar o executável

cd /workspaces/labsisop-buildroot
pgrep -a qemu
CC=$(ls output/host/bin/*buildroot*-gcc | head -n1)
$CC -O2 -pthread sched_profiler.c -o sched_profiler
file sched_profiler
e2cp -p sched_profiler output/images/rootfs.ext2:/root/sched_profiler
e2ls -l output/images/rootfs.ext2:/root

2. Testar no host (opcional)

gcc -O2 -pthread sched_profiler.c -o sched_profiler_host
taskset -c 0 ./sched_profiler_host 10000 4 SCHED_IDLE
sudo taskset -c 0 ./sched_profiler_host 10000 4 SCHED_RR

3. Iniciar QEMU

cd /workspaces/labsisop-buildroot
./output/images/start-qemu.sh serial-only

4. Testar no QEMU

cd /root
./sched_profiler 10000 4 SCHED_RR

5. Gravar traces no QEMU

trace-cmd record -o trace_other.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_OTHER > saida_other.txt"
awk '/Apos pos/{f=1} f' saida_other.txt

trace-cmd record -o trace_fifo.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_FIFO > saida_fifo.txt"
awk '/Apos pos/{f=1} f' saida_fifo.txt

trace-cmd record -o trace_rr.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_RR > saida_rr.txt"
awk '/Apos pos/{f=1} f' saida_rr.txt

trace-cmd record -o trace_idle.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_IDLE > saida_idle.txt"
awk '/Apos pos/{f=1} f' saida_idle.txt

trace-cmd record -o trace_low_idle.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_LOW_IDLE > saida_low_idle.txt"
awk '/Apos pos/{f=1} f' saida_low_idle.txt

trace-cmd record -o trace_misto.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 2 SCHED_LOW_IDLE,SCHED_IDLE > saida_misto.txt"
awk '/Apos pos/{f=1} f' saida_misto.txt

sync
poweroff

6. Copiar traces para o Codespace

cd /workspaces/labsisop-buildroot
for p in other fifo rr idle low_idle misto; do
  e2cp output/images/rootfs.ext2:/root/trace_$p.dat trace_$p.dat
  e2cp output/images/rootfs.ext2:/root/saida_$p.txt saida_$p.txt
done
ls -lh trace_*.dat saida_*.txt

7. Conferir traces

sudo apt-get update
sudo apt-get install trace-cmd -y
trace-cmd report -i trace_other.dat | head -20
trace-cmd report -i trace_fifo.dat | head -20
trace-cmd report -i trace_rr.dat | head -20
trace-cmd report -i trace_idle.dat | head -20
trace-cmd report -i trace_low_idle.dat | head -20
trace-cmd report -i trace_misto.dat | head -20

8. Iniciar KernelShark

curl -fsSL https://gist.githubusercontent.com/miguelxvr/3c123ef67611652af9d2e45eccb8fc0a/raw/ | bash -s -- start

9. Abrir traces no KernelShark

DISPLAY=:99 kernelshark /workspaces/labsisop-buildroot/trace_other.dat &
DISPLAY=:99 kernelshark /workspaces/labsisop-buildroot/trace_fifo.dat &
DISPLAY=:99 kernelshark /workspaces/labsisop-buildroot/trace_rr.dat &
DISPLAY=:99 kernelshark /workspaces/labsisop-buildroot/trace_idle.dat &
DISPLAY=:99 kernelshark /workspaces/labsisop-buildroot/trace_low_idle.dat &
DISPLAY=:99 kernelshark /workspaces/labsisop-buildroot/trace_misto.dat &

10. Testar com 2 CPUs (opcional)

cd /workspaces/labsisop-buildroot
sed 's/-M pc/-M pc -smp cpus=2/' output/images/start-qemu.sh > output/images/start-qemu-smp.sh
chmod +x output/images/start-qemu-smp.sh
./output/images/start-qemu-smp.sh serial-only

11. Gravar traces com 2 CPUs

cd /root
trace-cmd record -o trace_rr_2cpu.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_RR > saida_rr_2cpu.txt"
trace-cmd record -o trace_fifo_2cpu.dat -e sched:sched_switch sh -c "./sched_profiler 5000000 4 SCHED_FIFO > saida_fifo_2cpu.txt"
sync
poweroff

12. Copiar traces de 2 CPUs

cd /workspaces/labsisop-buildroot
e2cp output/images/rootfs.ext2:/root/trace_rr_2cpu.dat trace_rr_2cpu.dat
e2cp output/images/rootfs.ext2:/root/saida_rr_2cpu.txt saida_rr_2cpu.txt
e2cp output/images/rootfs.ext2:/root/trace_fifo_2cpu.dat trace_fifo_2cpu.dat
e2cp output/images/rootfs.ext2:/root/saida_fifo_2cpu.txt saida_fifo_2cpu.txt
ls -lh trace_*2cpu.dat saida_*2cpu.txt

13. Parar KernelShark

curl -fsSL https://gist.githubusercontent.com/miguelxvr/3c123ef67611652af9d2e45eccb8fc0a/raw/ | bash -s -- stop
